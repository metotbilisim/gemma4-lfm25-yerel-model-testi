            // The canonicality check ensures that the input string is exactly
            // what the toRoman method would produce for the parsed integer.
            // This handles non-canonical forms like "IIII", "VV", "IC", etc.
            // It also handles cases where the parsed integer is out of range.
            if (self::toRoman($total) !== $s) {
                throw new \InvalidArgumentException("Non-canonical Roman numeral.");
            }

            return $total;
        }

        public static function add(string $a, string $b): string
        {
            $valA = self::fromRoman($a);
            $valB = self::fromRoman($b);
            $sum = $valA + $valB;

            return self::toRoman($sum);
        }
