    ```php
    final class RomanNumerals
    {
        /**
         * @throws \InvalidArgumentException
         */
        public static function toRoman(int $n): string
        {
            if ($n < 1 || $n > 3999) {
                throw new \InvalidArgumentException("Number must be between 1 and 3999.");
            }

            $map = [
                'M' => 1000, 'CM' => 900, 'D' => 500, 'CD' => 400,
                'C' => 100, 'XC' => 90, 'L' => 50, 'XL' => 40,
                'X' => 10, 'IX' => 9, 'V' => 5, 'IV' => 4, 'I' => 1
            ];

            $result = '';
            foreach ($map as $roman => $value) {
                while ($n >= $value) {
                    $result .= $roman;
                    $n -= $value;
                }
            }

            return $result;
        }

        /**
         * @throws \InvalidArgumentException
         */
        public static function fromRoman(string $s): int
        {
            if ($s === '' || !preg_match('/^[IVXLCDM]+$/', $s)) {
                throw new \InvalidArgumentException("Invalid Roman numeral string.");
            }

            $map = ['I' => 1, 'V' => 5, 'X' => 10, 'L' => 50, 'C' => 100, 'D' => 500, 'M' => 1000];
            $total = 0;
            $len = strlen($s);

            for ($i = 0; $i < $len; $i++) {
                $current = $map[$s[$i]];
                $next = ($i + 1 < $len) ? $map[$s[$i + 1]] : 0;

                if ($current < $next) {
                    $total -= $current;
                } else {
                    $total += $current;
                }
            }

            // The canonical check: convert back and compare.
            // This handles non-canonical forms like "IIII", "VV", "IC", etc.
            // It also handles numbers out of range because toRoman will throw.
            if (self::toRoman($total) !== $s) {
                throw new \InvalidArgumentException("Non-canonical Roman numeral.");
            }

            return $total;
        }

        /**
         * @throws \InvalidArgumentException
         */
        public static function add(string $a, string $b): string
        {
            $valA = self::fromRoman($a);
            $valB = self::fromRoman($b);

            return self::toRoman($valA + $valB);
        }
    }
    ```
