    final class RomanNumerals
    {
        private const MAP = [
            'M' => 1000,
            'CM' => 900,
            'D' => 500,
            'CD' => 400,
            'C' => 100,
            'XC' => 90,
            'L' => 50,
            'XL' => 40,
            'X' => 10,
            'IX' => 9,
            'V' => 5,
            'IV' => 4,
            'I' => 1,
        ];

        private const REVERSE_MAP = [
            'I' => 1,
            'V' => 5,
            'X' => 10,
            'L' => 50,
            'C' => 100,
            'D' => 500,
            'M' => 1000,
        ];

        public static function toRoman(int $n): string
        {
            if ($n < 1 || $n > 3999) {
                throw new \InvalidArgumentException("Number out of range (1-3999).");
            }

            $result = '';
            foreach (self::MAP as $roman => $value) {
                while ($n >= $value) {
                    $result .= $roman;
                    $n -= $value;
                }
            }

            return $result;
        }

        public static function fromRoman(string $s): int
        {
            // Regex for canonical Roman numerals 1-3999
            // ^(?=[MDCLXVI]) ensures the string is not empty and contains valid characters
            // M{0,3} - Thousands
            // (CM|CD|D?C{0,3}) - Hundreds
            // (XC|XL|L?X{0,3}) - Tens
            // (IX|IV|V?I{0,3}) - Ones
            if (!preg_match('/^(?=[MDCLXVI])M{0,3}(CM|CD|D?C{0,3})(XC|XL|L?X{0,3})(IX|IV|V?I{0,3})$/', $s)) {
                throw new \InvalidArgumentException("Invalid or non-canonical Roman numeral.");
            }

            $total = 0;
            $length = strlen($s);
            for ($i = 0; $i < $length; $i++) {
                $currentVal = self::REVERSE_MAP[$s[$i]];
                $nextVal = ($i + 1 < $length) ? self::REVERSE_MAP[$s[$i + 1]] : 0;

                if ($currentVal < $nextVal) {
                    $total -= $currentVal;
                } else {
                    $total += $currentVal;
                }
            }

            return $total;
        }

        public static function add(string $a, string $b): string
        {
            $valA = self::fromRoman($a);
            $valB = self::fromRoman($b);
            $sum = $valA + $valB;

            return self::toRoman($sum);
        }
    }
