        public function set(string $key, mixed $value, ?float $ttl = null): void {
            if ($ttl !== null && $ttl <= 0) {
                throw new \InvalidArgumentException("TTL must be positive");
            }

            $now = ($this->clock)();
            $expiry = ($ttl !== null) ? ($now + $ttl) : null;

            if (isset($this->map[$key])) {
                $node = $this->map[$key];
                $node->value = $value;
                $node->expiry = $expiry;
                $this->moveToTail($node);
            } else {
                $node = new Node($key, $value, $expiry);
                $this->map[$key] = $node;
                $this->addToTail($node);
            }

            if (count($this->map) > $this->capacity) {
                // 1. Remove expired
                $curr = $this->head;
                while ($curr !== null) {
                    $next = $curr->next;
                    if ($curr->expiry !== null && $now >= $curr->expiry) {
                        $this->removeNode($curr);
                        unset($this->map[$curr->key]);
                    }
                    $curr = $next;
                }

                // 2. If still over, evict LRU
                if (count($this->map) > $this->capacity) {
                    if ($this->head !== null) {
                        $lru = $this->head;
                        $this->removeNode($lru);
                        unset($this->map[$lru->key]);
                    }
                }
            }
        }
