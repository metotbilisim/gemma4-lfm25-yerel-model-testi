    private class Parser {
        private readonly Lexer _lexer;
        private Token _currentToken;

        public Parser(Lexer lexer) {
            _lexer = lexer;
            _currentToken = _lexer.NextToken();
        }

        public Token PeekToken() => _currentToken;

        private void Eat(TokenType type) {
            if (_currentToken.Type == type) {
                _currentToken = _lexer.NextToken();
            } else {
                throw new FormatException($"Expected {type} but found {_currentToken.Type}");
            }
        }

        public double ParseExpression() {
            double left = ParseTerm();
            while (_currentToken.Type == TokenType.Plus || _currentToken.Type == TokenType.Minus) {
                TokenType op = _currentToken.Type;
                Eat(op);
                double right = ParseTerm();
                if (op == TokenType.Plus) left += right;
                else left -= right;
            }
            return left;
        }

        public double ParseTerm() {
            double left = ParseUnary();
            while (_currentToken.Type == TokenType.Multiply || _currentToken.Type == TokenType.Divide) {
                TokenType op = _currentToken.Type;
                Eat(op);
                double right = ParseUnary();
                if (op == TokenType.Multiply) left *= right;
                else {
                    if (right == 0) throw new DivideByZeroException();
                    left /= right;
                }
            }
            return left;
        }

        public double ParseUnary() {
            if (_currentToken.Type == TokenType.Plus) {
                Eat(TokenType.Plus);
                return ParseUnary();
            }
            if (_currentToken.Type == TokenType.Minus) {
                Eat(TokenType.Minus);
                return -ParseUnary();
            }
            return ParsePower();
        }

        public double ParsePower() {
            double left = ParsePrimary();
            if (_currentToken.Type == TokenType.Power) {
                Eat(TokenType.Power);
                double right = ParsePower(); // Right-associative
                return Math.Pow(left, right);
            }
            return left;
        }

        public double ParsePrimary() {
            if (_currentToken.Type == TokenType.Number) {
                double val = _currentToken.Value;
                Eat(TokenType.Number);
                return val;
            }
            if (_currentToken.Type == TokenType.LParen) {
                Eat(TokenType.LParen);
                double val = ParseExpression();
                Eat(TokenType.RParen);
                return val;
            }
            throw new FormatException($"Unexpected token: {_currentToken.Type}");
        }
    }

    One last check: `2^3^2` with `Power` -> `Primary [ ^ Unary ]`.
    Is it possible to have `2^3^2^4`?
    Yes, it will be `2^(3^(2^4))`.

    Wait, there's one potential issue. `2^3*4`.
    `ParsePower` -> `Primary(2)`, `Eat(^)`, `ParseUnary`
    `ParseUnary` -> `ParsePower` -> `Primary(3)`
    `ParsePower` returns `3`.
    `ParseUnary` returns `3`.
    `ParsePower` returns `2^3 = 8`.
    `ParseTerm` -> `8`, `Eat(*)`, `ParseUnary` -> `ParsePower` -> `Primary(4)`
    Result `8*4 = 32`.
    Correct.

    What about `2*3^4`?
    `ParseTerm` -> `ParseUnary(2)`, `Eat(*)`, `ParseUnary`
    `ParseUnary` -> `ParsePower` -> `Primary(3)`, `Eat(^)`, `ParseUnary` -> `ParsePower` -> `Primary(4)`
    Result `2*(3^4) = 162`.
    Correct.

    Final Grammar:
    `Expression` -> `Term` { (`+` | `-`) `Term` }
    `Term` -> `Unary` { (`*` | `/`) `Unary` }
    `Unary` -> (`+` | `-`) `Unary` | `Power`
    `Power` -> `Primary` [ `^` `Unary` ]
    `Primary` -> `Number` | `(` `Expression` `)`

    Wait, `Power` -> `Primary [ ^ Unary ]` is right-associative *only if* `Unary` calls `Power`.
    And `Unary` *does* call `Power`.
    So it's all good.

    *   `2^-1` -> `2 ^ (-1)`
    *   `2^3^2` -> `2 ^ (3 ^ 2)`
    *   `-2^2` -> `-(2^2)`
    *   `2*-3` -> `2 * (-3)`
    *   `--2` -> `-(-2)`
    *   `2^-(-1)` -> `2 ^ (-(-1))`

    All good.

    Wait, one more: `2^3^2` is `2^(3^2)`.
    My `ParsePower` is:
    ```csharp
    public double ParsePower() {
        double left = ParsePrimary();
        if (_currentToken.Type == TokenType.Power) {
            Eat(TokenType.Power);
            double right = ParseUnary(); // This is the key!
            return Math.Pow(left, right);
        }
        return left;
    }
    ```
    If `right = ParseUnary()`, then `2^3^2` will be `2^(3^2)`.
    Let's trace:
    `ParsePower` (1) calls `ParsePrimary` (2) -> `2`.
    `ParsePower` (1) sees `^`.
    `ParsePower` (1) calls `ParseUnary` (3).
    `ParseUnary` (3) calls `ParsePower` (4).
    `ParsePower` (4) calls `ParsePrimary` (5) -> `3`.
    `ParsePower` (4) sees `^`.
    `ParsePower` (4) calls `ParseUnary` (6).
    `ParseUnary` (6) calls `ParsePower` (7).
    `ParsePower` (7) calls `ParsePrimary` (8) -> `2`.
    `ParsePower` (7) returns `2`.
    `ParseUnary` (6) returns `2`.
    `ParsePower` (4) returns `Math.Pow(3, 2) = 9`.
    `ParseUnary` (3) returns `9`.
    `ParsePower` (1) returns `Math.Pow(2, 9) = 512`.
    Perfect.

    *   `2^3^2` -> `512`
    *   `2^-1` -> `0.5`
    *   `-2^2` -> `-4`
    *   `2*-3` -> `-6`
