private function addToTail(Node $node): void {
    // Insert after tail
    if ($this->tail === null) {
        $this->head = $this->tail = $node;
    } else {
        $this->tail->next = $node;
        $node->prev = $this->tail;
        $node->next = null;
        $this->tail = $node;
    }
}

private function moveToTail(Node $node): void {
    // If node already at tail? We'll just unlink and re-add.
    if ($node === $this->tail) return;
    // Unlink
    $node->prev->next = $node->next;
    if ($node->next !== null) {
        $node->next->prev = $node->prev;
    } else {
        // node was head
        $this->head = $node->next;
    }
    // Add to tail
    $this->addToTail($node);
}

private function removeNode(Node $node): void {
    // Unlink from list
    if ($node->prev !== null) {
        $node->prev->next = $node->next;
    } else {
        $this->head = $node->next;
    }
    if ($node->next !== null) {
        $node->next->prev = $node->prev;
    } else {
        $this->tail = $node->prev;
    }
    $node->prev = null;
    $node->next = null;
}

        $currentTime = $this->getClock();
        $count = 0;
        foreach ($this->map as $node) {
            if ($node->expiry > $currentTime) {
                $count++;
            }
        }
        return $count;
    }

    public function set(string $key, mixed $value, ?float $ttl = null): void {
        $currentTime = $this->getClock();

        if (isset($this->map[$key])) {
            $node = &$this->map[$key];
            $node->value = $value;
            if ($ttl !== null) {
                if ($ttl <= 0) {
                    throw new \InvalidArgumentException('TTL must be greater than zero');
                }
                $node->expiry = $currentTime + $ttl;
            } else {
                $node->expiry = PHP_INT_MAX;
            }
            $this->moveToTail($node);
        } else {
            if ($ttl !== null) {
                if ($ttl <= 0) {
                    throw new \InvalidArgumentException('TTL must be greater than zero');
                }
                $expiry = $currentTime + $ttl;
            } else {
                $expiry = PHP_INT_MAX;
            }
            $node = new Node($key, $value, $expiry);
            $this->map[$key] = $node;
            $this->addToTail($node);
        }

        // Enforce capacity
        $this->enforceCapacity($currentTime);
    }

    public function get(string $key, mixed $default = null): mixed {
        $currentTime = $this->getClock();
        if (!isset($this->map[$key])) {
            return $default;
        }
        $node = &$this->map[$key];
        if ($node->expiry <= $currentTime) {
            // Expired: remove
            $this->removeNode($node);
            unset($this->map[$key]);
            return $default;
        }
        // Not expired: move to MRU
        $this->moveToTail($node);
        return $node->value;
    }

    public function has(string $key): bool {
        $currentTime = $this->getClock();
        if (!isset($this->map[$key])) {
            return false;
        }
        $node = $this->map[$key];
        if ($node->expiry <= $currentTime) {
            return false;
        }
        return true;
    }

    public function delete(string $key): bool {
        $currentTime = $this->getClock();
        if (!isset($this->map[$key])) {
            return false;
        }
        $node = &$this->map[$key];
        if ($node->expiry <= $currentTime) {
            // Expired entry: treat as missing
            return false;
        }
        $this->removeNode($node);
        unset($this->map[$key]);
        return true;
    }

    public function count(): int {
        return $this->countNonExpired();
    }

    public function keys(): array {
        $currentTime = $this->getClock();
        $result = [];
        $node = $this->head;
        while ($node !== null) {
            if ($node->expiry > $currentTime) {
                $result[] = $node->key;
            }
            $node = $node->next;
        }
        return $result;
    }
