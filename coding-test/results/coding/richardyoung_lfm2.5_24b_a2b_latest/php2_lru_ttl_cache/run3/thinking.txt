class LruCache {
    private int $capacity;
    private ?Closure $clock;
    private array $entries = []; // key => Node
    private ?Node $head = null;
    private ?Node $tail = null;

    public function __construct(int $capacity, ?Closure $clock = null) {
        if ($capacity < 1) {
            throw new \InvalidArgumentException('Capacity must be at least 1');
        }
        $this->capacity = $capacity;
        $this->clock = $clock ?? null; // store as Closure|null
    }

    private function currentTime(): float {
        if ($this->clock !== null) {
            return $this->clock();
        }
        return microtime(true);
    }

    // linked list helpers
    private function addToTail(Node $node): void {
        if ($this->tail === null) {
            $this->head = $this->tail = $node;
        } else {
            $this->tail->next = $node;
            $node->prev = $this->tail;
            $this->tail = $node;
        }
    }

    private function removeNode(Node $node): void {
        if ($node->prev !== null) {
            $node->prev->next = $node->next;
        } else {
            $this->head = $node->next;
        }
        if ($node->next !== null) {
            $node->next->prev = $node->prev;
        } else {
            $this->tail = $node->prev;
        }
    }

    private function moveToTail(Node $node): void {
        // If already at tail, do nothing
        if ($node->next === null) {
            return;
        }
        // Detach from current position
        $this->removeNode($node);
        // Add to tail
        $this->addToTail($node);
    }

    private function isExpired(Node $node): bool {
        if ($node->expireAt === null) {
            return false;
        }
        return $this->currentTime() >= $node->expireAt;
    }

    private function cleanupExpired(): void {
        // Iterate over a copy of keys? Since we may delete nodes, need to be careful.
        // We'll traverse from head to tail.
        $curr = $this->head;
        while ($curr !== null) {
            $next = $curr->next; // save next before possibly deleting
            if ($this->isExpired($curr)) {
                // Remove node
                $this->removeNode($curr);
                unset($this->entries[$curr->key]);
                // Continue; no need to advance curr because we removed it.
            } else {
                $curr = $next;
            }
        }
    }

    private function enforceCapacity(): void {
        while ($this->count() > $this->capacity) {
            // Evict LRU (head)
            if ($this->head === null) break; // shouldn't happen
            $lruNode = $this->head;
            // Remove from list and entries
            $this->removeNode($lruNode);
            unset($this->entries[$lruNode->key]);
        }
    }

    public function set(string $key, mixed $value, ?float $ttl = null): void {
        if ($this->capacity < 1) {
            throw new \InvalidArgumentException('Capacity must be at least 1');
        }
        $time = $this->currentTime();
        if ($ttl !== null && $ttl <= 0) {
            throw new \InvalidArgumentException('TTL must be positive');
        }
        $expireAt = ($ttl !== null) ? $time + $ttl : null;

        // Check existing
        $existingNode = $this->entries[$key] ?? null;
        if ($existingNode !== null) {
            // Update value and expiration
            $existingNode->value = $value;
            $existingNode->expireAt = $expireAt;
            // Move to tail (most recent)
            $this->moveToTail($existingNode);
        } else {
            // Create new node
            $node = new Node($key, $value, $expireAt);
            $this->entries[$key] = $node;
            $this->addToTail($node);
        }

        // After insertion, clean up expired entries first
        $this->cleanupExpired();
        // Then enforce capacity
        $this->enforceCapacity();
    }

    public function get(string $key, mixed $default = null): mixed {
        $node = $this->entries[$key] ?? null;
        if ($node === null) {
            return $default;
        }
        $time = $this->currentTime();
        if ($node->expireAt !== null && $time >= $node->expireAt) {
            // Expired: remove
            $this->delete($key);
            return $default;
        }
        // Not expired: make recent
        $this->moveToTail($node);
        return $node->value;
    }

    public function has(string $key): bool {
        $node = $this->entries[$key] ?? null;
        if ($node === null) {
            return false;
        }
        $time = $this->currentTime();
        if ($node->expireAt !== null && $time >= $node->expireAt) {
            // Expired: remove
            $this->delete($key);
            return false;
        }
        return true;
    }

    public function delete(string $key): bool {
        $node = $this->entries[$key] ?? null;
        if ($node === null) {
            return false;
        }
        // Remove from list and array
        $this->removeNode($node);
        unset($this->entries[$key]);
        return true;
    }

    public function count(): int {
        // Count non-expired entries.
        // We'll iterate over entries (or linked list) and check expiration.
        // Since expired entries may still be present, we should count only those not expired.
        $count = 0;
        $curr = $this->head;
        while ($curr !== null) {
            if (!$this->isExpired($curr)) {
                $count++;
            } else {
                // Optionally remove? Not required. But to keep count accurate, we could delete it.
                // However spec for count: number of non-expired entries. If expired ones remain, they shouldn't be counted.
                // We'll just skip them; but they may linger. That's okay.
            }
            $curr = $curr->next;
        }
        return $count;
    }

    public function keys(): array {
        $keys = [];
        $curr = $this->head;
        while ($curr !== null) {
            if (!$this->isExpired($curr)) {
                $keys[] = $curr->key;
            }
            $curr = $curr->next;
        }
        return $keys;
    }
}

class LruCache {
    private int $capacity;
    private ?Closure $clock;
    private array $entries = [];
    private ?Node $head = null;
    private ?Node $tail = null;

    private class Node {
        public string $key;
        public mixed $value;
        public ?float $expireAt;
        public ?Node $prev;
        public ?Node $next;

        public function __construct(string $key, mixed $value, ?float $expireAt) {
            $this->key = $key;
            $this->value = $value;
            $this->expireAt = $expireAt;
            $this->prev = null;
            $this->next = null;
        }
    }

    public function __construct(int $capacity, ?Closure $clock = null) {
        if ($capacity < 1) {
            throw new \InvalidArgumentException('Capacity must be at least 1');
        }
        $this->capacity = $capacity;
        $this->clock = $clock;
    }

    private function currentTime(): float {
        return $this->clock !== null ? $this->clock() : microtime(true);
    }

    // linked list operations
    private function addToTail(Node $node): void {
        if ($this->tail === null) {
            $this->head = $this->tail = $node;
        } else {
            $this->tail->next = $node;
            $node->prev = $this->tail;
            $this->tail = $node;
        }
    }

    private function removeNode(Node $node): void {
        if ($node->prev !== null) {
            $node->prev->next = $node->next;
        } else {
            $this->head = $node->next;
        }
        if ($node->next !== null) {
            $node->next->prev = $node->prev;
        } else {
            $this->tail = $node->prev;
        }
    }

    private function moveToTail(Node $node): void {
        if ($node->next === null) {
            return;
        }
        $this->removeNode($node);
        $this->addToTail($node);
    }

    private function isExpired(Node $node): bool {
        if ($node->expireAt === null) {
            return false;
        }
        return $this->currentTime() >= $node->expireAt;
    }

    private function cleanupExpired(): void {
        $curr = $this->head;
        while ($curr !== null) {
            $next = $curr->next;
            if ($this->isExpired($curr)) {
                $this->removeNode($curr);
                unset($this->entries[$curr->key]);
            } else {
                $curr = $next;
            }
        }
    }

    private function enforceCapacity(): void {
        while ($this->count() > $this->capacity) {
            if ($this->head === null) {
                break;
            }
            $lruNode = $this->head;
            $this->removeNode($lruNode);
            unset($this->entries[$lruNode->key]);
        }
    }

    public function set(string $key, mixed $value, ?float $ttl = null): void {
        if ($this->capacity < 1) {
            throw new \InvalidArgumentException('Capacity must be at least 1');
        }
        $time = $this->currentTime();
        if ($ttl !== null && $ttl <= 0) {
            throw new \InvalidArgumentException('TTL must be positive');
        }
        $expireAt = ($ttl !== null) ? $time + $ttl : null;

        $existingNode = $this->entries[$key] ?? null;
        if ($existingNode !== null) {
            // Update existing entry
            $existingNode->value = $value;
            $existingNode->expireAt = $expireAt;
            $this->moveToTail($existingNode);
        } else {
            // Insert new entry
            $node = new Node($key, $value, $expireAt);
            $this->entries[$key] = $node;
            $this->addToTail($node);
        }

        // Remove expired entries first
        $this->cleanupExpired();
        // Enforce capacity
        $this->enforceCapacity();
    }

    public function get(string $key, mixed $default = null): mixed {
        $node = $this->entries[$key] ?? null;
        if ($node === null) {
            return $default;
        }
        if ($this->isExpired($node)) {
            $this->delete($key);
            return $default;
        }
        // Not expired: make most recently used
        $this->moveToTail($node);
        return $node->value;
    }

    public function has(string $key): bool {
        $node = $this->entries[$key] ?? null;
        if ($node === null) {
            return false;
        }
        if ($this->isExpired($node)) {
            $this->delete($key);
            return false;
        }
        return true;
    }

    public function delete(string $key): bool {
        $node = $this->entries[$key] ?? null;
        if ($node === null) {
            return false;
        }
        $this->removeNode($node);
        unset($this->entries[$key]);
        return true;
    }

    public function count(): int {
        $count = 0;
        $curr = $this->head;
        while ($curr !== null) {
            if (!$this->isExpired($curr)) {
                $count++;
            }
            $curr = $curr->next;
        }
        return $count;
    }

    public function keys(): array {
        $keys = [];
        $curr = $this->head;
        while ($curr !== null) {
            if (!$this->isExpired($curr)) {
                $keys[] = $curr->key;
            }
            $curr = $curr->next;
        }
        return $keys;
    }
