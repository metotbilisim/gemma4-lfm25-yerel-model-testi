'use strict';

async function promisePool(taskFns, limit) {
    if (!Number.isInteger(limit) || limit <= 0) {
        return Promise.reject(new TypeError('limit must be a positive integer'));
    }

    const n = taskFns.length;
    const results = new Array(n);
    let nextIdx = 0;
    let activeCount = 0;
    let hasError = false;
    let errorVal = null;

    return new Promise((resolve, reject) => {
        async function runTask(idx) {
            try {
                const ret = await taskFns[idx]();
                results[idx] = ret;
                return ret;
            } catch (e) {
                results[idx] = e;
                errorVal = e;
                hasError = true;
                reject(e);
                throw e;
            }
        }

        function startNext() {
            if (hasError || nextIdx >= n || activeCount >= limit) return;
            const idx = nextIdx++;
            activeCount++;
            runTask(idx)
                .then(() => {
                    activeCount--;
                    if (nextIdx < n && activeCount < limit && !hasError) {
                        startNext();
                    }
                })
                .catch(() => {
                    activeCount--;
                });
        }

        // Start initial batch
        for (let i = 0; i < Math.min(n, limit); i++) {
            startNext();
        }

        // Wait until all tasks have finished
        const timer = setInterval(() => {
            if (activeCount === 0 && nextIdx >= n) {
                clearInterval(timer);
                if (!hasError) {
                    resolve(results);
                }
                // If there was an error the outer promise has already been rejected.
            }
        }, 10);

        // Immediate resolution for empty task list
        if (n === 0) {
            clearInterval(timer);
            resolve(results);
        }
    });
}

async function retry(fn, options = {}) {
    const { retries = 3, delayMs = 0, factor = 2, shouldRetry = (err) => true } = options;
    let attempt = 1;
    let lastError = null;

    while (true) {
        try {
            const result = await fn(attempt);
            return result;
        } catch (e) {
            lastError = e;
            if (attempt > retries || !shouldRetry(e)) {
                throw e;
            }
            const wait = delayMs * Math.pow(factor, attempt - 1);
            await new Promise(resolve => setTimeout(resolve, wait));
            attempt++;
        }
    }
}

module.exports = { promisePool, retry };
